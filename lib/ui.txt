function createWindow()
{
    return new window();
}

function createImageFromFile(file)
{
    return new image(file);
}

class image
{
    var imageId;

    function image(fileName)
    {
        imageId = intern readImage(fileName);
    }
}

class canvas
{
    var canvasId;

    function canvas(id)
    {
        canvasId = id;
    }

    function drawText(text, x, y)
    {
        intern canvasDrawText(canvasId, text, x, y);
    }

    function drawImage(image, x, y, width, height)
    {
        intern canvasDrawImage(canvasId, image, x, y, width, height);
    }
}

class window
{
    var windowId;

    function window()
    {
        windowId = intern windowCreate();
    }

    function setVisible(value)
    {
        intern windowSetVisible(windowId, value);
    }

    function setSize(width, height)
    {
        intern windowSetSize(windowId, width, height);
    }

    function setName(name)
    {
        intern windowSetName(windowId, name);
    }

    function setDrawHandler(drawHandler)
    {
        intern windowSetDrawHandler(windowId, drawHandler);
    }

    function centerOnScreen()
    {
        intern windowCenterOnScreen(windowId);
    }
}
